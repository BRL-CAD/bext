cmake_minimum_required(VERSION 3.26)
project(DeflatePackageCheck LANGUAGES CXX)

if(NOT libdeflate_ROOT)
  message(FATAL_ERROR "Set libdeflate_ROOT to the installed package prefix to validate")
endif()
find_package(libdeflate CONFIG REQUIRED)
file(REAL_PATH "${libdeflate_ROOT}" expected_root)
foreach(target libdeflate::libdeflate_shared libdeflate::libdeflate_static)
  if(NOT TARGET ${target})
    continue()
  endif()
  get_target_property(includes ${target} INTERFACE_INCLUDE_DIRECTORIES)
  foreach(directory IN LISTS includes)
    file(REAL_PATH "${directory}" include_path)
    cmake_path(IS_PREFIX expected_root "${include_path}" NORMALIZE in_prefix)
    if(NOT in_prefix OR NOT IS_DIRECTORY "${include_path}")
      message(FATAL_ERROR "Invalid installed include directory for ${target}: ${directory}")
    endif()
  endforeach()
endforeach()

add_executable(deflate_package_check main.cpp)
target_compile_features(deflate_package_check PRIVATE cxx_std_17)
target_link_libraries(deflate_package_check PRIVATE libdeflate::libdeflate_shared)
enable_testing()
add_test(NAME deflate_package COMMAND deflate_package_check)
