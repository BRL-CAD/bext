cmake_minimum_required(VERSION 3.26)
project(IcuDataCheck LANGUAGES CXX)

if(NOT ICU_ROOT)
  message(FATAL_ERROR "Set ICU_ROOT to the installed ICU prefix to validate")
endif()
find_package(ICU REQUIRED COMPONENTS uc data)
file(REAL_PATH "${ICU_ROOT}" expected_root)
foreach(library IN LISTS ICU_LIBRARIES)
  if(library STREQUAL "optimized" OR library STREQUAL "debug" OR library STREQUAL "general")
    continue()
  endif()
  file(REAL_PATH "${library}" library_path)
  cmake_path(IS_PREFIX expected_root "${library_path}" NORMALIZE in_prefix)
  if(NOT in_prefix)
    message(FATAL_ERROR "ICU library is outside the requested prefix: ${library}")
  endif()
endforeach()

foreach(directory IN LISTS ICU_INCLUDE_DIRS)
  file(REAL_PATH "${directory}" include_path)
  cmake_path(IS_PREFIX expected_root "${include_path}" NORMALIZE in_prefix)
  if(NOT in_prefix OR NOT IS_DIRECTORY "${include_path}")
    message(FATAL_ERROR "ICU include directory is outside the requested prefix: ${directory}")
  endif()
endforeach()

add_executable(icu_data_check main.cpp)
target_compile_features(icu_data_check PRIVATE cxx_std_17)
target_link_libraries(icu_data_check PRIVATE ICU::uc ICU::data)
enable_testing()
add_test(NAME icu_data COMMAND icu_data_check)
set_tests_properties(icu_data PROPERTIES ENVIRONMENT "ICU_DATA=")
