# Decision made

if (ENABLE_OPENVDB)

  git_submodule_init(openvdb CMakeLists.txt)

  RegisterDeps(OPENVDB)
  if (TARGET ZLIB_BLD)
    set(Z_PREFIX_STR "brl_")
  endif (TARGET ZLIB_BLD)

  ExternalProject_Add(OPENVDB_BLD
    URL "${CMAKE_CURRENT_SOURCE_DIR}/openvdb"
    BUILD_ALWAYS ${EXT_BUILD_ALWAYS} ${LOG_OPTS}
    PATCH_COMMAND ${CMAKE_COMMAND};-E;copy;${PROJECT_SOURCE_DIR}/CMake/FindZLIB.cmake;<SOURCE_DIR>/cmake/FindZLIB.cmake
    CMAKE_ARGS
    -DCMAKE_FIND_PACKAGE_PREFER_CONFIG=ON
    -DMIN_VER_CMAKE=3.19
    ${BUILD_TYPE_SPECIFIER}
    # Do not let an earlier system-zlib result override the bundled ZLIB_ROOT.
    $<$<BOOL:${ZLIB_TARGET}>:-UZLIB_INCLUDE_DIR>
    $<$<BOOL:${ZLIB_TARGET}>:-UZLIB_LIBRARY*>
    $<$<BOOL:${ZLIB_TARGET}>:-DZ_PREFIX=ON>
    $<$<BOOL:${ZLIB_TARGET}>:-DZ_PREFIX_STR=${Z_PREFIX_STR}>
    $<$<BOOL:${ZLIB_TARGET}>:-DZLIB_ROOT=${CMAKE_BUNDLE_INSTALL_PREFIX}>
    $<$<BOOL:${ONETBB_TARGET}>:-DTBB_ROOT=${CMAKE_BUNDLE_INSTALL_PREFIX}>
    -DBIN_DIR=${BIN_DIR}
    -DLIB_DIR=${LIB_DIR}
    -DCMAKE_INSTALL_LIBDIR:PATH=${LIB_DIR}
    -DBUILD_STATIC_LIBS=${BUILD_STATIC_LIBS}
    -DCMAKE_CXX_COMPILER=${CMAKE_CXX_COMPILER}
    -DCMAKE_C_COMPILER=${CMAKE_C_COMPILER}
    -DCMAKE_INSTALL_PREFIX=${CMAKE_BUNDLE_INSTALL_PREFIX}
    -DCMAKE_INSTALL_LIBDIR:PATH=${LIB_DIR}
    -DCMAKE_INSTALL_RPATH=${CMAKE_BUNDLE_INSTALL_PREFIX}/${LIB_DIR}
    -DUSE_PKGCONFIG=OFF
    -DOPENVDB_BUILD_BINARIES=OFF
    -DOPENVDB_BUILD_UNITTESTS=OFF
    -DOPENVDB_ENABLE_VEC_RELATIONAL_OPERATIONS=OFF
    -DOPENVDB_INSTALL_CMAKE_MODULES=ON
    -DOPENVDB_ENABLE_UNINSTALL=OFF
    -DOPENVDB_USE_DELAYED_LOADING=OFF
    -DOPENVDB_X86_INSTRSET=6
    -DOPENVDB_DISABLE_BOOST_IMPLICIT_LINKING=ON
    -DUSE_BLOSC=OFF
    -DUSE_EXR=OFF
    -DUSE_IMATH_HALF=OFF
    -DUSE_LOG4CPLUS=OFF
    -DUSE_PNG=OFF
    LOG_CONFIGURE ${EXT_BUILD_QUIET}
    LOG_BUILD ${EXT_BUILD_QUIET}
    LOG_INSTALL ${EXT_BUILD_QUIET}
    LOG_OUTPUT_ON_FAILURE ${EXT_BUILD_QUIET}
    STEP_TARGETS install
    )

  TargetInstallDeps(OPENVDB OPENVDB_DEPENDS)

  # Copy the license into position in CMAKE_BUNDLE_INSTALL_PREFIX
  configure_file(
    ${CMAKE_CURRENT_SOURCE_DIR}/openvdb/LICENSE
    ${DOC_LICENSE_DIR}/openvdb.txt
    COPYONLY
    )

endif (ENABLE_OPENVDB)
